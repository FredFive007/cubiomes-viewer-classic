cmake_minimum_required(VERSION 3.21)
project(CubiomesViewer263Classic LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_EXTENSIONS ON)
find_package(Qt5 5.15 REQUIRED COMPONENTS Core Widgets)
file(GLOB ENGINE_CORE CONFIGURE_DEPENDS cubiomes/*.c)
list(FILTER ENGINE_CORE EXCLUDE REGEX "/tests\\.c$")
file(GLOB_RECURSE ENGINE_EXTRA CONFIGURE_DEPENDS cubiomes/features/*.c cubiomes/loot/*.c)
add_library(cubiomes_engine STATIC ${ENGINE_CORE} ${ENGINE_EXTRA})
target_compile_options(cubiomes_engine PRIVATE -fwrapv -fno-semantic-interposition)
target_compile_definitions(cubiomes_engine PRIVATE STRUCT_CONFIG_OVERRIDE=1)
file(GLOB VIEWER_CPP CONFIGURE_DEPENDS src/*.cpp)
list(FILTER VIEWER_CPP EXCLUDE REGEX "/updater\\.cpp$")
file(GLOB VIEWER_HEADERS CONFIGURE_DEPENDS src/*.h)
file(GLOB VIEWER_FORMS CONFIGURE_DEPENDS src/*.ui)
file(GLOB LUA_C CONFIGURE_DEPENDS lua/src/*.c)
list(FILTER LUA_C EXCLUDE REGEX "/lua(c)?\\.c$")
add_executable(cubiomes-viewer-26.3-classic WIN32 ${VIEWER_CPP} ${VIEWER_HEADERS} ${VIEWER_FORMS}
    ${LUA_C} src/libwinsane/init.c src/libwinsane/manifest.rc
    rc/icons.qrc rc/style.qrc rc/examples.qrc rc/lang.qrc)
set_target_properties(cubiomes-viewer-26.3-classic PROPERTIES AUTOMOC ON AUTOUIC ON AUTORCC ON)
target_include_directories(cubiomes-viewer-26.3-classic PRIVATE . src src/libwinsane)
target_compile_definitions(cubiomes-viewer-26.3-classic PRIVATE STRUCT_CONFIG_OVERRIDE=1 QT_DISABLE_DEPRECATED_UP_TO=0x050F00)
target_compile_options(cubiomes-viewer-26.3-classic PRIVATE -fwrapv -finput-charset=UTF-8 -fexec-charset=UTF-8)
target_link_libraries(cubiomes-viewer-26.3-classic PRIVATE cubiomes_engine Qt5::Core Qt5::Widgets)

include(CTest)
if(BUILD_TESTING)
    foreach(name biomes structures loot pieces carvers)
        add_executable(test_${name} cubiomes/tests/test_${name}.c)
        target_link_libraries(test_${name} PRIVATE cubiomes_engine)
        add_test(NAME engine_${name} COMMAND test_${name})
    endforeach()
    add_test(NAME viewer_port COMMAND cubiomes-viewer-26.3-classic
        --verify-port ${CMAKE_CURRENT_BINARY_DIR}/port-verification)
    set_tests_properties(viewer_port PROPERTIES TIMEOUT 60)
    add_executable(test_vanilla validation/test_vanilla.c)
    target_include_directories(test_vanilla PRIVATE .)
    target_link_libraries(test_vanilla PRIVATE cubiomes_engine)
    add_test(NAME vanilla_26_3_reference COMMAND test_vanilla
        ${CMAKE_CURRENT_SOURCE_DIR}/validation/vanilla-26.3-biomes.csv)
endif()
